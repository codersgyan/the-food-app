== nginx config (/etc/nginx/sites-enabled/food-api.sys0.in.conf)
# nginx in front of our API server, as in the reverse proxy lesson.
#
# nginx answers HTTPS on port 443 for food-api.sys0.in and passes each request
# to our API server on port 8080 of the same machine, over plain HTTP. The API
# server listens only on 127.0.0.1, so nobody on the internet can reach it
# without going through nginx.
#
# Two things are left out of the lesson's last config, because our app
# doesn't have them yet: the 5 MB size limit for photos of dishes, and the
# restaurant dashboard at /dashboard/. nginx's default limit of 1 MB is plenty
# for our requests.
#
# Install it as /etc/nginx/sites-available/food-api.sys0.in.conf.
# deploy/README.md has every step.

# Plain HTTP only sends the client to HTTPS. Customers send their addresses
# and phone numbers, so nothing is answered over plain HTTP.
server {
    listen 80;
    listen [::]:80;
    server_name food-api.sys0.in;

    location / {
        return 301 https://$host$request_uri;
    }
}

server {
    listen 443 ssl;
    listen [::]:443 ssl;
    server_name food-api.sys0.in;

    # Compress JSON too. By default nginx compresses only HTML.
    gzip on;
    gzip_types application/json;

    # Requests from the customer's app and the restaurant's tablet.
    location /v1/ {
        proxy_pass http://127.0.0.1:8080;
    }

    # The payment provider's webhook. It isn't under /v1/, so it needs its own
    # location, or the provider's requests would never reach our server.
    location /payments/ {
        proxy_pass http://127.0.0.1:8080;
    }

    # Anything else isn't ours, so answer 404 instead of nginx's welcome page.
    location / {
        return 404;
    }

    # Let's Encrypt's certificate for the domain. certbot creates these files
    # and renews them before they expire.
    ssl_certificate     /etc/letsencrypt/live/food-api.sys0.in/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/food-api.sys0.in/privkey.pem;
}
